## js实现
```js
function merge(nums, temp, left, mid, right) {
  let l = left;
  let r = mid + 1;
  let k = l;
  while (l <= mid && r <= right) {
    temp[k++] = nums[l] <= nums[r] ? nums[l++] : nums[r++];
  }

  while (l <= mid) temp[k++] = nums[l++];
  while (r <= right) temp[k++] = nums[r++];
  for (let i = left; i <= right; i++) nums[i] = temp[i];
}

function handleSort(nums) {
  if (nums.length < 2) return;

  let temp = Array(nums.length);
  function mergeSort(l, r) {
    if (l < r) {
      let mid = l + Math.floor((r - l) / 2);
      mergeSort(l, mid);
      mergeSort(mid + 1, r);
      if (nums[mid] > nums[mid + 1]) merge(nums, temp, l, mid, r);
    }
  }
  mergeSort(0, nums.length - 1);
}
```
