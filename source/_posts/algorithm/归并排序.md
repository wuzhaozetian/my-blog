## js实现
```js
function mergeSort(nums) {
  let temp = Array(nums.length);

  function merge(left, mid, right) {
    let l = left;
    let r = mid + 1;
    let k = l;
    while (l <= mid && r <= right) {
      temp[k++] = nums[l] <= nums[r] ? nums[l++] : nums[r++];
    }

    while (l <= mid) temp[k++] = nums[l++];
    while (r <= right) temp[k++] = nums[r++];

    for (let i = left; i <= right; i++) nums[i] = temp[i];
  }

  function divide(l, r) {
    if (l < r) {
      let mid = Math.floor((r - l) / 2) + l;
      divide(l, mid);
      divide(mid + 1, r);
      if (nums[mid] > nums[mid + 1]) merge(l, mid, r);
    }
  }
  divide(0, nums.length - 1);
}
```
