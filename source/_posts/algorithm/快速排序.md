## js实现
```js
function quickSort(nums) {
  //挖坑法 一边找比基准元素小的值，一边找比基准元素大的值
  function quick(left, right) {
    let pivot = nums[left];
    let l = left;
    let r = right;

    while (l < r) {
      while (l < r && nums[r] >= pivot) r--;
      nums[l] = nums[r];
      while (l < r && nums[l] <= pivot) l++;
      nums[r] = nums[l];
    }
    nums[l] = pivot;
    return l;
  }

  function divide(l, r) {
    if (l < r) {
      let mid = quick(l, r);
      divide(l, mid - 1);
      divide(mid + 1, r);
    }
  }
  divide(0, nums.length - 1);
}

function quickSort2(nums) {
  //Lomuto分区法的变体 只找比基准元素小的值
  function quick(left, right) {
    let pivot = nums[left];
    let first = left + 1;
    for (let i = first; i <= right; i++) {
      if (pivot > nums[i]) {
        [nums[i], nums[first]] = [nums[first], nums[i]];
        first++;
      }
    }
    let ans = first - 1;
    [nums[left], nums[ans]] = [nums[ans], nums[left]];
    return ans;
  }
  function divide(l, r) {
    if (l < r) {
      let mid = quick(l, r);
      divide(l, mid - 1);
      divide(mid + 1, r);
    }
  }
  divide(0, nums.length - 1);
}
```
