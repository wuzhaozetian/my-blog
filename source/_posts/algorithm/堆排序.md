## js实现
```js
function heapSort(nums) {
  //前提条件是：左子树是最大堆，右子树是最大堆。
  //构建最大堆
  function heap(parent, end) {
    let l = 2 * parent + 1;
    let r = 2 * parent + 2;

    let max = parent;
    if (l < end && nums[l] > nums[max]) max = l;
    if (r < end && nums[r] > nums[max]) max = r;
    if (max != parent) {
      [nums[max], nums[parent]] = [nums[parent], nums[max]];
      heap(max, end);
    }
  }

  let temp = Math.floor(nums.length / 2) - 1;
  for (let i = temp; i > -1; i--) {
    heap(i, nums.length);
  }

  for (let i = nums.length - 1; i > 0; i--) {
    [nums[0], nums[i]] = [nums[i], nums[0]];
    heap(0, i);
  }
}
```
